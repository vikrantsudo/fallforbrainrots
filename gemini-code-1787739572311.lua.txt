-- ===================== DRAGGING =====================
local function makeDraggable(handle, frameToMove)
	local dragging, dragStart, startPos, dragInput
	handle.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true; dragStart = input.Position; startPos = frameToMove.Position
			input.Changed:Connect(function() if input.UserInputState == Enum.UserInputState.End then dragging = false end end)
		end
	end)
	handle.InputChanged:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then dragInput = input end
	end)
	UserInputService.InputChanged:Connect(function(input)
		if input == dragInput and dragging then
			local delta = input.Position - dragStart
			frameToMove.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
		end
	end)
end

local function createUnifiedToggle(parent, defaultState, callback)
	local toggle = Instance.new("TextButton", parent)
	toggle.Size = UDim2.new(0, 42, 0, 22)
	toggle.Position = UDim2.new(1, -47, 0.5, -11)
	toggle.BackgroundColor3 = defaultState and Color3.fromRGB(220, 35, 35) or Color3.fromRGB(30, 30, 35)
	toggle.Text = defaultState and "[ON]" or "[OFF]"
	toggle.TextColor3 = defaultState and Color3.fromRGB(255, 255, 255) or Color3.fromRGB(160, 160, 160)
	toggle.Font = Enum.Font.Code; toggle.TextSize = 10
	Instance.new("UICorner", toggle).CornerRadius = UDim.new(0, 2)
	local state = defaultState or false
	toggle.MouseButton1Click:Connect(function()
		state = not state
		toggle.Text = state and "[ON]" or "[OFF]"
		toggle.BackgroundColor3 = state and Color3.fromRGB(220, 35, 35) or Color3.fromRGB(30, 30, 35)
		toggle.TextColor3 = state and Color3.fromRGB(255, 255, 255) or Color3.fromRGB(160, 160, 160)
		if callback then callback(state) end
	end)
	return toggle
end

local EspEnabled, FlyEnabled, SpeedEnabled, NoclipEnabled, TracersEnabled, GravityEnabled, GodModeEnabled, XrayEnabled = false, false, false, false, false, false, false, false
local FlySpeed, WalkSpeedValue = 50, 32
local EspObjects, TracerLines, xrayCache = {}, {}, {}

-- ===================== CORE LOOPS & SYSTEMS =====================
RunService.RenderStepped:Connect(function(dt)
	-- ESP & Tracers Loop
	for _, p in ipairs(Players:GetPlayers()) do
		if p ~= LocalPlayer and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
			local hrp = p.Character.HumanoidRootPart
			local pos, onScreen = Camera:WorldToViewportPoint(hrp.Position)
			-- ESP
			if EspEnabled and onScreen then
				if not EspObjects[p] then
					local box = Drawing.new("Square"); box.Color = Color3.fromRGB(220, 35, 35); box.Thickness = 1; box.Filled = false
					local name = Drawing.new("Text"); name.Color = Color3.fromRGB(255, 255, 255); name.Size = 13; name.Center = true; name.Outline = true
					EspObjects[p] = {Box = box, Name = name}
				end
				local size = Vector2.new(38, 54)
				EspObjects[p].Box.Size, EspObjects[p].Box.Position, EspObjects[p].Box.Visible = size, Vector2.new(pos.X - size.X / 2, pos.Y - size.Y / 2), true
				EspObjects[p].Name.Text, EspObjects[p].Name.Position, EspObjects[p].Name.Visible = p.Name, Vector2.new(pos.X, pos.Y - size.Y / 2 - 14), true
			elseif EspObjects[p] then EspObjects[p].Box.Visible, EspObjects[p].Name.Visible = false, false end
			
			-- Tracers
			if TracersEnabled and onScreen then
				if not TracerLines[p] then
					local line = Drawing.new("Line"); line.Color = Color3.fromRGB(220, 35, 35); line.Thickness = 1; TracerLines[p] = line
				end
				TracerLines[p].From, TracerLines[p].To, TracerLines[p].Visible = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y), Vector2.new(pos.X, pos.Y), true
			elseif TracerLines[p] then TracerLines[p].Visible = false end
		end
	end

	-- True CFRAME FLY 
	if FlyEnabled and LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then
		local hrp = LocalPlayer.Character.HumanoidRootPart
		local move = Vector3.zero
		if UserInputService:IsKeyDown(Enum.KeyCode.W) then move = move + Camera.CFrame.LookVector end
		if UserInputService:IsKeyDown(Enum.KeyCode.S) then move = move - Camera.CFrame.LookVector end
		if UserInputService:IsKeyDown(Enum.KeyCode.A) then move = move - Camera.CFrame.RightVector end
		if UserInputService:IsKeyDown(Enum.KeyCode.D) then move = move + Camera.CFrame.RightVector end
		if UserInputService:IsKeyDown(Enum.KeyCode.Space) then move = move + Vector3.new(0, 1, 0) end
		if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) or UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then move = move - Vector3.new(0, 1, 0) end
		hrp.AssemblyLinearVelocity = Vector3.zero
		hrp.AssemblyAngularVelocity = Vector3.zero
		if move.Magnitude > 0 then hrp.CFrame = hrp.CFrame + (move.Unit * (FlySpeed * dt * 3)) end
	end

	-- Speed Engine
	if SpeedEnabled and LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Humanoid") then
		LocalPlayer.Character.Humanoid.WalkSpeed = WalkSpeedValue
	end
    
    -- God Mode Constant Loop
    if GodModeEnabled and LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Humanoid") then
        LocalPlayer.Character.Humanoid.MaxHealth = math.huge
        LocalPlayer.Character.Humanoid.Health = math.huge
    end
end)

	if NoclipConn then NoclipConn:Disconnect() NoclipConn = nil end

		NoclipConn = RunService.Stepped:Connect(function()

				end
			end
		end)
	end
end

-- ===================== SEARCH & TABS =====================
local currentTab = "Cmds"
local function applySearch(query)
	local lq = string.lower(query)
	if currentTab == "Cmds" then
		for _, c in ipairs(moduleCards) do c.Box.Visible = (lq == "" or string.find(string.lower(c.Name), lq, 1, true) ~= nil) end
	else
		for _, b in ipairs(EmotesContainer:GetChildren()) do if b:IsA("TextButton") then b.Visible = (lq == "" or string.find(string.lower(b.Text), lq, 1, true) ~= nil) end end
	end
end
TabCmds.MouseButton1Click:Connect(function() currentTab = "Cmds"; CmdsContainer.Visible = true; EmotesContainer.Visible = false; TabCmds.BackgroundColor3 = Color3.fromRGB(220, 35, 35); TabEmotes.BackgroundColor3 = Color3.fromRGB(30, 30, 36); applySearch(SearchBox.Text) end)
TabEmotes.MouseButton1Click:Connect(function() currentTab = "Emotes"; CmdsContainer.Visible = false; EmotesContainer.Visible = true; TabEmotes.BackgroundColor3 = Color3.fromRGB(220, 35, 35); TabCmds.BackgroundColor3 = Color3.fromRGB(30, 30, 36); applySearch(SearchBox.Text) end)
SearchBox:GetPropertyChangedSignal("Text"):Connect(function() applySearch(SearchBox.Text) end)
